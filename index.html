// ===================================================================
// Safari history.replaceState rate-limit guard
// Safari enforces a 100-calls-per-10s cap on history.replaceState /
// history.pushState and throws SecurityError on call 101+. Our router
// (svelte-spa-router's push()) is async -- the throw becomes a rejected
// promise that aborts navigation before window.location.hash is set.
// We swallow ONLY the rate-limit SecurityError so the async caller
// continues past the throw and the hash update still runs. All other
// SecurityErrors (cross-origin, unsupported URL) still propagate.
// Registered inline BEFORE the Dub analytics script so it sits directly
// on top of the native history API. Third-party scripts that wrap
// history.* later will wrap our wrapper, and our catch still runs.
// See docs/common-pitfalls.md -> "Safari history.replaceState rate limit".
// ===================================================================
(function wrapHistoryForSafariRateLimit() {
if (typeof history === "undefined") return;
var RATE_LIMIT_RE = /more than 100 times per 10 seconds/i;
function isRateLimit(err) {
return !!err &&
err.name === "SecurityError" &&
RATE_LIMIT_RE.test(String(err.message || ""));
}
["replaceState", "pushState"].forEach(function(fn) {
var orig = history[fn];
if (typeof orig !== "function") return;
history[fn] = function() {
try {
return orig.apply(this, arguments);
} catch (err) {
if (isRateLimit(err)) {
// Safari dropped the history state save. App navigation
// continues via window.location.hash; the scroll-state
// piggyback on this call is lost but not critical.
return;
}
throw err;
}
};
});
})();

// ===================================================================
// Critical Error Bootstrap
// Runs inline before any JS chunks load. Handles the case where the
// main JS bundle fails to load (deployment race, R2 upload delay,
// stale service worker serving stale HTML with old chunk refs).
//
// Without this, chunk load failures are completely invisible:
// - The app silently stalls on the splash screen
// - No error is reported to Loki (error code is in the failed chunk)
// - No auto-reload is triggered (that logic is also in the failed chunk)
//
// This script and main.ts share the same sessionStorage key for the
// 30s reload cooldown, so they cannot double-reload each other.
// ===================================================================
(function() {
var CHUNK_RELOAD_KEY = "chipp-chunk-reload";
var CHUNK_RELOAD_COOLDOWN_MS = 30000;
var ERROR_ENDPOINT = "/api/client-errors";

// Stores the last pre-boot error so the stall beacon can include it.
window.__chippBootError = null;
// Boot stage set by main.ts and App.svelte for stall diagnostics.
// undefined -> entry module not yet evaluated
// "entry-evaluated" -> main.ts started, IIFEs running
// "mounted" -> mount() returned, Svelte initializing
// "app-booted" -> App.svelte onMount ran, splash cleared
window.__chippBootStage = window.__chippBootStage || undefined;

// CHIPPD-2481: detect browsers that cannot execute <script type="module">
// AT ALL (Chrome <61, Safari <10.1, Firefox <60 -- includes legacy in-app
// WebViews such as the Android System WebView shipped with Chrome 55 seen
// in production boot-stall telemetry). These devices still fetch the entry
// bundle (module preload/prefetch happens independently of execution
// support) but the module script can never run -- this is exactly the
// `boot-stall-downloaded-not-executed` population in categorize.ts, and NO
// amount of waiting or reloading fixes it: the browser itself lacks the
// capability. Detected via the standard nomodule/module feature test
// (`noModule` exists on script elements only in browsers implementing the
// HTML module-script spec) -- synchronous, no network round trip needed.
// The splash-timeout watchdog in <body> reads this to skip the 8s wait and
// show an honest "update your browser" message instead of a useless
// "Tap to reload" button for this specific, unrecoverable population.
try {
window.__chippModuleUnsupported = !("noModule" in document.createElement("script"));
} catch (eModuleDetect) {
window.__chippModuleUnsupported = false;
}

// Reload the page once per 30 seconds to recover from chunk errors.
// Uses the same sessionStorage key as main.ts -- if main.ts already
// reloaded, the inline handler won't double-reload.
function tryChunkReload(source) {
try {
var last = parseInt(sessionStorage.getItem(CHUNK_RELOAD_KEY) || "0", 10);
if (Date.now() - last > CHUNK_RELOAD_COOLDOWN_MS) {
sessionStorage.setItem(CHUNK_RELOAD_KEY, String(Date.now()));
console.warn("[Chipp] " + source + " failed, reloading for new deployment...");
location.reload();
}
} catch (e) {
// sessionStorage unavailable (storage-partitioned/sandboxed
// iframe, private browsing). An unconditional reload here loops
// FOREVER on a persistent chunk failure (each generation fires a
// beacon, bounded only by the server rate limiter). Bound it to
// ONE reload per page generation via a URL marker instead.
// Kept in lockstep with main.ts tryChunkReload + share.html.
try {
if (location.search.indexOf("chipp_cr=1") === -1) {
var sep = location.search ? "&" : "?";
location.replace(
location.pathname + location.search + sep + "chipp_cr=1" + location.hash
);
}
} catch (e2) { /* give up quietly */ }
}
}

// Send a minimal error report to Loki without depending on any JS chunk.
// tags.source routes to the correct categorize.ts classification:
// "chunk_load_error" -> warn (chunk-load)
// "boot_error" / "boot_exception" -> log.error (boot-path)
// "boot_stall" -> log.error only when extra.bootStage shows our code
// ran (post-mount hang); else warn (module never evaluated / slow net)
// extra.handler = "inline-bootstrap" distinguishes these entries in Loki queries.
function sendError(name, message, source, extraData, stack) {
try {
var errObj = {
name: name || "ChunkLoadError",
message: String(message || "unknown error").slice(0, 2000),
url: location.href,
timestamp: new Date().toISOString(),
tags: { source: source || "chunk_load_error" },
extra: Object.assign({ handler: "inline-bootstrap" }, extraData || {})
};
if (stack) errObj.stack = String(stack).slice(0, 2000);
var body = JSON.stringify({ errors: [errObj] });
if (navigator.sendBeacon) {
navigator.sendBeacon(ERROR_ENDPOINT, new Blob([body], { type: "application/json" }));
} else {
fetch(ERROR_ENDPOINT, {
method: "POST",
headers: { "Content-Type": "application/json" },
body: body,
keepalive: true
}).catch(function() {});
}
} catch (e) {
// Error reporting must never throw
}
}

// Expose for the body splash-timeout watchdog: sendError is scoped to
// this IIFE, and the watchdog lives in a SEPARATE <script> block in
// <body>. Without this assignment its `typeof` guard silently resolves
// null and the boot_stall beacon never fires (the frozen-splash class
// becomes invisible in Loki). Regression-guarded by
// src/__tests__/web/boot_stall_beacon_wiring_test.ts.
window.__chippSendError = sendError;

// True once the app's own error handling is live. Two signals, in
// trust order:
// 1. __chippClientErrorsReady -- ErrorBoundary.svelte attached its
// window handlers (set in its onMount; ErrorBoundary is the
// unconditional top-level wrapper in App.svelte, and its onMount
// fires before App.svelte's). This closes the former dead zone

// the old "#app has children" check flipped this off at first
// DOM insertion, so a synchronous error thrown mid-mount (e.g.
// in a child component's onMount) was reported by NEITHER layer.
// 2. __chippBootStage === "app-booted" -- App.svelte finished boot.
//
// Deliberately NO "#app has children" fallback: index.html and its

// zone it used to cause.
function appHasBooted() {
if (window.__chippClientErrorsReady) return true;
return window.__chippBootStage === "app-booted";
}

// Diagnostics for chunk-load beacons: which asset failed, was it the
// boot-blocking entry doc (assets/main-*.js / assets/share-*.js --
// every visitor mid-load is stuck, urgent) or a lazy dynamic-import
// route chunk (normal, self-healing deploy churn), and (best-effort)

// implementation in web/src/lib/utils/chunk-diagnostics.ts -- kept as

var ENTRY_CHUNK_RE = /^(?:main|share)-[A-Za-z0-9_-]{8}\.js$/;
var ENTRY_CHUNK_LEGACY_RE = /^(?:main|share)\.[a-f0-9]{8}\.js$/;
function chunkBasename(pathOrUrl) {
var noQuery = String(pathOrUrl || "").split(/[?#]/)[0];
var parts = noQuery.split("/");
return parts[parts.length - 1] || noQuery;
}
function chunkDiagnostics(assetUrl) {
var diag = {};
if (!assetUrl) return diag;
var base = chunkBasename(assetUrl);
diag.chunkKind = (ENTRY_CHUNK_RE.test(base) || ENTRY_CHUNK_LEGACY_RE.test(base))
? "entry"
: "dynamic";
try {
if (typeof performance !== "undefined" && typeof performance.getEntriesByName === "function") {
var entries = performance.getEntriesByName(assetUrl);
var status = entries[0] && entries[0].responseStatus;
if (typeof status === "number" && status > 0) diag.httpStatus = status;
}
} catch (e) { /* Resource Timing L2 unsupported -- status stays unknown */ }
